def bfs_nostart(g, start):
    q = deque([start])
    visited = set()          # 시작 노드를 넣지 않음
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in g[u]:
            if v not in visited:
                visited.add(v)
                q.append(v)
    return order

print(bfs_nostart(F, "P"))
