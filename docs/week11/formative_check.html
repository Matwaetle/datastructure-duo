Q3 · visited 함정

아래 코드를 F에서 시작 P로 실행했을 때 출력되는 방문 순서를 쓰고, 중복이 생기는 이유와 고치는 방법을 한 문장씩 쓰시오.

from collections import deque

# F: 인접 리스트(Q2와 같은 그래프). 이웃 순서는 목록 순서이다.
F = {"P": ["Q", "R"], "Q": ["P", "S"], "R": ["P", "S", "T"], "S": ["Q", "R", "U"], "T": ["R", "U"], "U": ["S", "T"]}

def bfs_nostart(g, start):                   # 시작 상태: 큐 [start], visited는 비어 있다
    q = deque([start])
    visited = set()

    while q:
        u = q.popleft()
        order.append(u)
        for v in g[u]:
            if v not in visited:
                visited.add(v)
                q.append(v)
    return order

print(bfs_nostart(F, "P"))
